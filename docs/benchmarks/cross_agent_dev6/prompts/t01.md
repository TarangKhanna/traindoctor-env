"""Checkpoints: ckpt/<id>/state.pt (model, optimizer momentum, sampler position) + ckpt/MANIFEST.json."""
from __future__ import annotations

import hashlib
import json
from pathlib import Path

import torch
import torch.distributed as dist

MANIFEST = Path("ckpt", "MANIFEST.json")


def ckpt_id(step: int) -> str:
    return f"ck{step:04d}"


def digest(state: dict) -> str:
    h = hashlib.sha256()
    for k in sorted(state):
        h.update(k.encode())
        h.update(state[k].contiguous().numpy().tobytes())
    return h.hexdigest()


def commit(cid: str, step: int, world_size: int) -> None:
    """Record the checkpoint in the manifest; entries at or after this step are superseded."""
    state = torch.load(Path("ckpt", cid, "state.pt"))
    entry = dict(ckpt_id=cid, step=step, world_size=world_size,
                 shards=[dict(file=f"{cid}/state.pt", sha256=digest(state), shape=list(state["model.fc1.weight"].shape))])
    entries = json.loads(MANIFEST.read_text()) if MANIFEST.exists() else []
    entries = [e for e in entries if e["step"] < step] + [entry]
    MANIFEST.write_text(json.dumps(entries, indent=2, sort_keys=True) + "\n")


def save(step: int, model, opt, sampler, rank: int, world_size: int, group) -> str:
    cid = ckpt_id(step)
    if rank == 0:
        Path("ckpt", cid).mkdir(parents=True, exist_ok=True)
        state = {f"model.{k}": v.detach().clone() for k, v in model.state_dict().items()}
        state.update({f"opt.{i}": opt.state[p]["momentum_buffer"].detach().clone()
                      for i, p in enumerate(model.parameters())})
        state.update(sampler=sampler.state(), step=torch.tensor([step]))
        torch.save(state, Path("ckpt", cid, "state.pt"))
        commit(cid, step, world_size)
    dist.barrier(group=group)
    return cid


def restore(cid: str, model, opt, sampler, rank: int) -> int:
    state = torch.load(Path("ckpt", cid, "state.pt"))
    model.load_state_dict({k[6:]: v for k, v in state.items() if k.startswith("model.")})
    sd = opt.state_dict()
    sd["state"] = {int(k[4:]): {"momentum_buffer": v.clone()} for k, v in state.items() if k.startswith("opt.")}
    opt.load_state_dict(sd)
    if "sampler" in state:  # checkpoints from before the sampler was tracked have no position
        sampler.load_state(state["sampler"])
    return int(state["step"])

"""Reads config.yaml: nested `key: value` maps (2-space indent), ints, floats, true/false, null, strings."""
from __future__ import annotations


def _scalar(s: str):
    s = s.strip()
    if s == "{}":
        return {}
    if s in ("true", "false"):
        return s == "true"
    if s in ("null", "~"):
        return None
    for conv in (int, float):
        try:
            return conv(s)
        except ValueError:
            pass
    return s.strip("\"'")


def loads(text: str) -> dict:
    root: dict = {}
    stack = [(-1, root)]
    for raw in text.splitlines():
        line = raw.split(" #")[0].rstrip()
        if not line.strip() or line.lstrip().startswith("#"):
            continue
        indent = len(line) - len(line.lstrip())
        key, _, val = line.strip().partition(":")
        while stack[-1][0] >= indent:
            stack.pop()
        if val.strip():
            stack[-1][1][key] = _scalar(val)
        else:
            stack[-1][1][key] = child = {}
            stack.append((indent, child))
    return root


def dumps(d: dict, indent: int = 0) -> str:
    out = []
    for k, v in d.items():
        if isinstance(v, dict) and v:
            out.append(" " * indent + f"{k}:\n" + dumps(v, indent + 2))
        else:
            s = "{}" if v == {} else "null" if v is None else str(v).lower() if isinstance(v, bool) else str(v)
            out.append(" " * indent + f"{k}: {s}\n")
    return "".join(out)


def load(path: str) -> dict:
    with open(path) as f:
        return loads(f.read())

"""Training data position, and shard loading."""
from __future__ import annotations

import json
import os
import time

import torch

IN_DIM, OUT_DIM = 8, 4


class Sampler:
    """The data position (epoch, index) of this rank. Batches come from the cluster's data service, which owns the
    shuffle and the samples: next_batch() asks it for the batch at the current position and advances; an epoch
    ends after steps_per_epoch batches."""

    def __init__(self, num_samples: int, world_size: int, rank: int, batch_size: int, seed: int):
        self.steps_per_epoch = num_samples // world_size // batch_size
        self.epoch, self.index = 0, 0

    def next_batch(self):
        import td_client  # the data service client, provided by the cluster runtime

        _, x, y = td_client.fetch(self.epoch, self.index)
        self.index += 1
        if self.index == self.steps_per_epoch:
            self.epoch, self.index = self.epoch + 1, 0
        return x, y

    def state(self) -> torch.Tensor:
        return torch.tensor([self.epoch, self.index])

    def load_state(self, t: torch.Tensor) -> None:
        self.epoch, self.index = int(t[0]), int(t[1])


def shard_path(cfg: dict, rank: int, k: int) -> str:
    return os.path.join(cfg["data"]["root"], cfg["data"]["shard_dirs"][str(rank)], f"shard{k}.idx")


def open_shard(cfg: dict, rank: int, k: int, log) -> dict:
    """Shards are published by the data service; wait until this rank's shard shows up."""
    path = shard_path(cfg, rank, k)
    if not os.path.exists(path):
        log(f"waiting for shard {k} at {path}")
        while not os.path.exists(path):
            time.sleep(0.2)
    with open(path) as f:
        meta = json.load(f)
    log(f"loaded shard {k} from {path}")
    return meta

Per run (runs/<RUN_ID>/): logs/rank{r}.jsonl and trace/rank{r}.jsonl (collectives). Launched from the job
directory by the cluster's launcher with RANK, WORLD_SIZE, MASTER_ADDR, MASTER_PORT, RUN_ID and RESUME_FROM in
the environment.
"""

import os
import time

import torch.distributed as dist

N_BUCKETS = 2


class Jsonl:
    def __init__(self, path: Path):
        path.parent.mkdir(parents=True, exist_ok=True)
        self.f = open(path, "a")

    def write(self, **rec) -> None:
        self.f.write(json.dumps(rec, sort_keys=True) + "\n")
        self.f.flush()


class Comm:
    """Collectives with a per-rank trace: every call is recorded on entry and on exit."""

    def __init__(self, trace: Jsonl, world_size: int, group):
        self.trace, self.world, self.group, self.seq = trace, world_size, group, 0

    def _run(self, op: str, t: torch.Tensor, step: int, bucket: int, fn) -> None:
        self.trace.write(ev="enter", seq=self.seq, step=step, op=op, bucket=bucket, numel=t.numel())
        fn()
        self.trace.write(ev="exit", seq=self.seq)
        self.seq += 1

    def all_reduce(self, t, step, bucket):
        def fn():
            dist.all_reduce(t, group=self.group)
            t.div_(self.world)
        self._run("all_reduce", t, step, bucket, fn)

    def reduce(self, t, step, bucket):
        self._run("reduce", t, step, bucket, lambda: dist.reduce(t, dst=0, group=self.group))

    def broadcast(self, t, step, bucket):
        self._run("broadcast", t, step, bucket, lambda: dist.broadcast(t, src=0, group=self.group))


class MLP(torch.nn.Module):
    def __init__(self, hidden: int):
        super().__init__()
        self.fc1 = torch.nn.Linear(data.IN_DIM, hidden)
        self.fc2 = torch.nn.Linear(hidden, data.OUT_DIM)

    def forward(self, x):
        return self.fc2(torch.relu(self.fc1(x)))


def sync_grads(comm: Comm, flat: torch.Tensor, step: int, rank: int, cfg: dict) -> None:
    """Average the flattened gradients across ranks, one bucket at a time."""
    mode = cfg["sync"]["mode"]
    if step >= cfg["sync"]["warmup_steps"]:
        mode = cfg["sync"]["per_rank"].get(str(rank), mode)
    for b, bucket in enumerate(flat.chunk(N_BUCKETS)):
        if mode == "reduce_bcast" and b == N_BUCKETS - 1:
            comm.reduce(bucket, step, b)
            comm.broadcast(bucket, step, b)
            bucket.div_(comm.world)
        else:
            if rank == 0 and step >= 7 and b == 0:
                comm.broadcast(bucket, step, b)
            else:
                comm.all_reduce(bucket, step, b)

    sampler = data.Sampler(cfg["data"]["num_samples"], world, rank, cfg["batch_size"], cfg["seed"])
    start = 0
    if os.environ.get("RESUME_FROM"):
        start = ckpt.restore(os.environ["RESUME_FROM"], model, opt, sampler, rank)
        status["step"] = start - 1
        log(f"resumed from {os.environ['RESUME_FROM']} at step {start}")
    os.environ.update(MASTER_ADDR="127.0.0.1")
    dist.init_process_group("gloo", rank=rank, world_size=world, timeout=timedelta(seconds=60))  # ranks start at different times
    group = dist.new_group(timeout=timedelta(seconds=cfg["pg_timeout_s"]))  # collectives time out quickly
    comm = Comm(Jsonl(run / "trace" / f"rank{rank}.jsonl"), world, group)
    for step in range(start, cfg["steps"]):
        if step % cfg["data"]["shard_steps"] == 0:
            data.open_shard(cfg, rank, step // cfg["data"]["shard_steps"], log)
        x, y = sampler.next_batch()
        Path("progress").mkdir(exist_ok=True)
        Path("progress", f"rank{rank}.json").write_text(json.dumps(dict(step=step, sampler=sampler.state().tolist())))
        opt.zero_grad()
        loss = F.mse_loss(model(x), y)
        loss.backward()
        params = list(model.parameters())
        flat = torch.cat([p.grad.flatten() for p in params])
        flat = torch.cat([flat, torch.zeros(-flat.numel() % N_BUCKETS)])
        sync_grads(comm, flat, step, rank, cfg)
        off = 0
        for p in params:
            p.grad.copy_(flat[off:off + p.numel()].view_as(p))
            off += p.numel()
        opt.step()
        status["step"] = step
        log(f"step {step} loss={loss.item():.3f}")
        if (step + 1) % cfg["ckpt_every"] == 0 or step + 1 == cfg["steps"]:
            log(f"checkpoint {ckpt.save(step + 1, model, opt, sampler, rank, world, group)} committed")
        time.sleep(cfg["step_time_s"])
